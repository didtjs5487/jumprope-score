구글 시트 연결 방법

한 번만 하면 됩니다(약 5분). 이후에는 앱 현황판의 구글 시트로 보내기 버튼만 누르면 같은 시트에 최신 성적이 덮어써집니다.

1빈 구글 시트 만들기

  1. 구글 드라이브에서 새로 만들기 → Google 스프레드시트를 누릅니다.
  2. 이름을 음악줄넘기 성적으로 바꿉니다.

2스크립트 붙여넣기

  1. 시트 위쪽 메뉴에서 확장 프로그램 → Apps Script를 엽니다.
  2. 가운데 편집창의 기존 글(function myFunction…)을 모두 지웁니다.
  3. 아래 스크립트 복사를 누르고, 편집창에 붙여넣습니다.
  4. 맨 위 KEY = '여기에-열쇠를-적으세요'의 따옴표 안을 선생님이 정한 암호로 바꿉니다. (예: 영문·숫자 8자 이상, 다른 사이트에 쓰는 비밀번호는 쓰지 마세요)
  5. 위쪽 저장(디스크 모양)을 누릅니다.
// 음악줄넘기 채점 → 구글 시트 받는 스크립트
// 사용법은 "구글시트_연결방법" 문서를 보세요.

// ▼ 여기 따옴표 안에 선생님이 정한 열쇠(암호)를 적으세요. 앱의 '열쇠'와 똑같아야 합니다.
const KEY = '여기에-열쇠를-적으세요';

function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    if (KEY === '여기에-열쇠를-적으세요') return out({ ok: false, error: '스크립트의 열쇠를 아직 정하지 않았습니다.' });
    if (d.key !== KEY) return out({ ok: false, error: '열쇠가 맞지 않습니다.' });

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sent = {};
    let people = 0;

    d.sheets.forEach(function (sh) {
      sent[sh.name] = true;
      const ws = ss.getSheetByName(sh.name) || ss.insertSheet(sh.name);
      ws.clear();
      const rows = sh.rows;
      if (!rows.length) return;
      const w = Math.max.apply(null, rows.map(function (r) { return r.length; }));
      const grid = rows.map(function (r) { const c = r.slice(); while (c.length < w) c.push(''); return c; });
      ws.getRange(1, 1, grid.length, w).setValues(grid);
      ws.getRange(1, 1, 1, w).setFontWeight('bold').setBackground('#EDF3FF');
      ws.setFrozenRows(1);
      ws.autoResizeColumns(1, w);
      if (sh.name === '전체') people = rows.length - 1;
    });

    // 이번에 기록이 없어진 반 시트는 옛 내용이 남지 않게 비워 둔다.
    ss.getSheets().forEach(function (ws) {
      if (/^\d+반$/.test(ws.getName()) && !sent[ws.getName()]) ws.clear();
    });

    // '전체' 시트를 맨 앞으로
    const all = ss.getSheetByName('전체');
    if (all) { ss.setActiveSheet(all); ss.moveActiveSheet(1); }

    // 맨 뒤 '업데이트' 시트에 마지막으로 받은 시각을 남긴다.
    const log = ss.getSheetByName('업데이트') || ss.insertSheet('업데이트');
    log.getRange('A1:B3').setValues([
      ['마지막 업데이트', Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm')],
      ['보낸 사람', d.by || ''],
      ['학생 수', people]
    ]);
    log.getRange('A1:A3').setFontWeight('bold');

    return out({ ok: true, rows: people, sheets: d.sheets.length });
  } catch (err) {
    return out({ ok: false, error: String(err) });
  }
}

// 앱의 '연결 확인' 버튼이 부르는 곳 (성적은 돌려주지 않고 시트 이름만 알려 줍니다)
function doGet(e) {
  return out({ ok: true, name: SpreadsheetApp.getActiveSpreadsheet().getName() });
}

function out(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

3웹 앱으로 배포하기

  1. 오른쪽 위 배포 → 새 배포를 누릅니다.
  2. 톱니바퀴(유형 선택) → 웹 앱을 고릅니다.
  3. 다음 사용자 인증 정보로 실행: 나 로 둡니다.
  4. 액세스 권한이 있는 사용자: 모든 사용자 로 바꿉니다.
  5. 배포를 누르면 권한 승인 창이 뜹니다. 권한 검토 → 계정 선택 → 고급 → (안전하지 않음) 이동 → 허용 순서로 누릅니다. (본인이 만든 스크립트라서 나오는 안내입니다.)
  6. 마지막 창에 웹 앱 URL이 나옵니다. 복사를 누릅니다. 주소는 https://script.google.com/macros/s/…/exec 로 끝납니다.
"모든 사용자"로 배포해도 암호를 모르면 아무것도 쓸 수 없습니다. 스크립트는 암호가 맞을 때만 시트에 씁니다. 그래서 암호는 남에게 알려 주지 마세요.

4앱에 연결하기

  1. 앱 명단 탭 → 구글 시트 연결 설정을 엽니다.
  2. 웹 앱 주소 칸에 3단계에서 복사한 주소, 열쇠 칸에 2단계에서 정한 암호를 넣습니다.
  3. 저장 후 연결 확인을 누릅니다. "연결됐습니다. 시트: 음악줄넘기 성적"이 나오면 성공입니다.
  4. 현황판의 구글 시트로 보내기를 눌러 보세요. 시트에 전체·반별 탭이 생기고 성적이 채워집니다.
앱을 쓰는 기기(휴대폰, 컴퓨터)마다 4단계를 한 번씩 해 주세요. 주소와 암호는 그 기기에만 저장됩니다.

5알아 둘 점

  • 버튼을 누를 때마다 시트의 전체·반별 탭이 새 성적으로 통째로 바뀝니다. 시트에서 직접 고친 내용은 사라지니, 계산이나 메모는 다른 탭에 적으세요.
  • 업데이트 탭에 마지막 업데이트 시각이 기록됩니다.
  • 스크립트를 고친 뒤에는 배포 → 배포 관리 → 새 버전으로 다시 배포해야 반영됩니다. 이때 주소는 그대로입니다.
  • 보강 선생님 계정은 연결이 필요 없습니다. 담당 선생님 기기에서만 보내시면 됩니다.
  • 시트는 선생님만 볼 수 있게 비공개로 두세요. 학생 번호·이름·점수가 들어 있습니다.

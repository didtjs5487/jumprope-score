구글 시트 연결 방법

한 번만 하면 됩니다(약 5분). 끝나면 성적이 바뀔 때마다 구글 드라이브의 스프레드시트가 자동으로 최신이 됩니다. 열쇠 파일 같은 어려운 설정은 없습니다.

1스프레드시트 준비

  1. 구글 드라이브의 음악줄넘기 성적 시트를 엽니다. (없으면 새로 만들기 → Google 스프레드시트로 만들고 이름을 같게 합니다.)
  2. 시트는 비공개로 두세요. 학생 번호·이름·점수가 들어갑니다.

2스크립트 붙여넣기

  1. 시트 위쪽 메뉴에서 확장 프로그램 → Apps Script를 엽니다.
  2. 가운데 편집창의 기존 글(function myFunction…)을 모두 지웁니다.
  3. 아래 스크립트 복사를 누르고 편집창에 붙여넣습니다.
  4. 맨 위 KEY = '여기에-암호를-적으세요'의 따옴표 안을 선생님이 정한 암호로 바꿉니다. (영문·숫자 8자 이상. 다른 곳에 쓰는 비밀번호는 쓰지 마세요.)
  5. 위쪽 저장(디스크 모양)을 누릅니다.
// 음악줄넘기 채점 → 구글 시트 받는 스크립트
// 앱이 성적이 바뀔 때마다 보내 주는 내용을 이 시트에 덮어씁니다.
// 사용법은 "구글시트_연결방법" 문서를 보세요. (열쇠 파일 같은 건 필요 없습니다)

// ▼ 따옴표 안에 선생님이 정한 암호를 적으세요. 앱의 '열쇠'와 똑같아야 합니다.
const KEY = '여기에-암호를-적으세요';

function doPost(e) {
  // 두 기기가 동시에 보내도 차례대로 처리합니다.
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(25000);
    const d = JSON.parse(e.postData.contents);
    if (KEY === '여기에-암호를-적으세요') { logRejected('스크립트의 KEY(암호)를 아직 정하지 않았습니다.'); return out({ ok: false }); }
    if (d.key !== KEY) { logRejected('암호가 맞지 않습니다. 앱의 열쇠와 스크립트의 KEY가 같은지 확인하세요.'); return out({ ok: false }); }
    const people = writeSheets(d.sheets, d.by || '');
    return out({ ok: true, rows: people, sheets: d.sheets.length });
  } catch (err) {
    try { logRejected('오류: ' + String(err)); } catch (e2) {}
    return out({ ok: false, error: String(err) });
  } finally {
    try { lock.releaseLock(); } catch (e3) {}
  }
}

// 앱의 '연결 확인' 버튼이 부르는 곳 (성적은 돌려주지 않고 시트 이름만 알려 줍니다)
function doGet(e) {
  return out({ ok: true, name: SpreadsheetApp.getActiveSpreadsheet().getName() });
}

function writeSheets(sheets, by) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sent = {};
  let people = 0;

  sheets.forEach(function (sh) {
    sent[sh.name] = true;
    const ws = ss.getSheetByName(sh.name) || ss.insertSheet(sh.name);
    ws.clear();
    const rows = sh.rows;
    if (!rows.length) return;
    const w = Math.max.apply(null, rows.map(function (r) { return r.length; }));
    const grid = rows.map(function (r) { const c = r.slice(); while (c.length < w) c.push(''); return c; });
    ws.getRange(1, 1, grid.length, w).setValues(grid);
    ws.getRange(1, 1, 1, w).setFontWeight('bold').setBackground('#EDF3FF');
    ws.setFrozenRows(1);
    ws.autoResizeColumns(1, w);
    if (sh.name === '전체') people = rows.length - 1;
  });

  // 이번에 기록이 없어진 반 시트는 옛 내용이 남지 않게 비워 둡니다.
  ss.getSheets().forEach(function (ws) {
    if (/^\d+반$/.test(ws.getName()) && !sent[ws.getName()]) ws.clear();
  });

  // 처음 만들어진 빈 '시트1'은 지우고, '전체'를 맨 앞으로
  const blank = ss.getSheetByName('시트1') || ss.getSheetByName('Sheet1');
  if (blank && ss.getSheets().length > 1 && blank.getLastRow() === 0) ss.deleteSheet(blank);
  const all = ss.getSheetByName('전체');
  if (all) { ss.setActiveSheet(all); ss.moveActiveSheet(1); }

  // '업데이트' 탭에 마지막으로 받은 시각과 보낸 사람을 남깁니다.
  const log = ss.getSheetByName('업데이트') || ss.insertSheet('업데이트');
  log.getRange('A1:B4').setValues([
    ['마지막 업데이트', Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss')],
    ['보낸 곳', by || ''],
    ['학생 수', people],
    ['결과', '성공']
  ]);
  log.getRange('A1:A4').setFontWeight('bold');
  log.autoResizeColumns(1, 2);
  return people;
}

// 앱은 구글의 응답을 읽을 수 없으므로, 거절한 이유를 '업데이트' 탭에 남깁니다.
function logRejected(reason) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const log = ss.getSheetByName('업데이트') || ss.insertSheet('업데이트');
  log.getRange('A6:B6').setValues([['마지막 거절', Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm') + ' · ' + reason]]);
  log.getRange('A6').setFontWeight('bold');
}

function out(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

3웹 앱으로 배포하기

  1. 오른쪽 위 배포 → 새 배포를 누릅니다.
  2. 톱니바퀴(유형 선택) → 웹 앱을 고릅니다.
  3. 다음 사용자 인증 정보로 실행: 나 로 둡니다.
  4. 액세스 권한이 있는 사용자: 모든 사용자 로 바꿉니다.
  5. 배포를 누르면 권한 승인 창이 뜹니다. 권한 검토 → 계정 선택 → 고급 → (안전하지 않음) 이동 → 허용 순서로 누릅니다. (본인이 만든 스크립트라서 나오는 안내입니다.)
  6. 마지막 창의 웹 앱 URL 옆 복사를 누릅니다. https://script.google.com/macros/s/…/exec 로 끝나는 주소입니다.
"모든 사용자"로 배포해도 2단계에서 정한 암호를 모르면 아무것도 쓸 수 없습니다. 그래서 암호는 남에게 알려 주지 마세요.

4앱에 연결하기

  1. 앱 명단 탭 → 구글 시트 연결 설정을 엽니다.
  2. 웹 앱 주소에 3단계에서 복사한 주소, 열쇠에 2단계에서 정한 암호를 넣고 저장을 누릅니다.
  3. 현황판의 구글 시트로 보내기를 눌러 시험합니다. 시트에 전체·반별 탭과 업데이트 탭이 생기면 성공입니다.
한 번 저장하면 선생님 계정으로 로그인한 다른 기기(휴대폰·컴퓨터)에도 자동으로 적용됩니다. 기기마다 다시 넣지 않아도 됩니다.

5알아 둘 점

  • 성적이 바뀐 뒤 약 20초 안에 시트에 반영됩니다. 여러 번 연달아 채점하면 모아서 한 번에 보내고, 같은 내용은 다시 보내지 않습니다.
  • 앱이 열려 있을 때 보냅니다. 성적은 앱으로만 바뀌므로, 채점하는 동안에는 항상 반영됩니다. 앱을 닫아 둔 상태에서 서버의 성적이 바뀐 경우(다른 선생님이 채점)는 그 선생님의 앱이 보냅니다.
  • 구글은 앱에 답을 돌려주지 않습니다. 그래서 앱은 "보냈습니다"까지만 알려 주고, 잘 반영됐는지와 오류(암호가 틀림 등)는 시트의 업데이트 탭에 적힙니다. 암호가 틀리면 "마지막 거절" 줄에 이유가 남습니다.
  • 성적이 지난번에 보낸 것보다 줄어들면(학생이 빠지거나 점수가 지워짐) 자동 전송은 시트를 바꾸지 않고, 앱에 이유를 띄웁니다. 점수를 고친 것처럼 개수가 같거나 늘어난 경우는 그대로 반영됩니다. 일부러 줄이신 거라면(새 학기 초기화 등) 현황판의 구글 시트로 보내기를 눌러 직접 보내세요.
  • 시트의 전체·반별 탭은 보낼 때마다 통째로 바뀝니다. 시트에서 직접 고친 내용은 사라지니 계산·메모는 다른 탭에 적으세요.
  • 자동 전송을 끄려면 명단 탭의 연결 설정에서 성적이 바뀔 때마다 자동으로 보내기를 끄고 저장하세요. 현황판 단추로는 언제든 직접 보낼 수 있습니다.
  • 스크립트를 고친 뒤에는 배포 → 배포 관리 → 새 버전으로 다시 배포해야 반영됩니다. 주소는 그대로입니다.
  • 새 학기에는 시트를 새로 복사해 쓰세요.
